import pygame import random import sys import math pygame.init() WIDTH, HEIGHT = 1000, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("RIFFAN.IO - 1923 EDITION") clock = pygame.time.Clock() # ألوان الريف BLACK = (10, 10, 10) GOLD = (212, 175, 55) RIF_RED = (180, 20, 20) WHITE = (255, 255, 255) AMAZIGH_BLUE = (0, 80, 160) font_big = pygame.font.SysFont('Arial', 48, bold=True) font_small = pygame.font.SysFont('Arial', 24, bold=True) # متغيرات العملة - هنا فين كتربط مشروعك TOKEN_NAME = "$RIFFAN" TOKEN_SUPPLY = 1923000000 # 1923 تاريخ الجمهورية coins_collected = 0 class RiffanCoin: def __init__(self): self.x = random.randint(WIDTH, WIDTH+500) self.y = random.randint(100, HEIGHT-100) self.size = 40 self.speed = 4 self.angle = 0 def update(self): self.x -= self.speed self.angle += 5 if self.x < -50: self.x = WIDTH + random.randint(0, 300) self.y = random.randint(100, HEIGHT-100) def draw(self): # رسم العملة ديالك الكحلة مع ₿ و ⵣ center = (int(self.x), int(self.y)) # دائرة كحلة pygame.draw.circle(screen, BLACK, center, self.size) # حاشية ذهبية pygame.draw.circle(screen, GOLD, center, self.size, 4) # رمز البيتكوين btc = font_small.render("₿", True, GOLD) screen.blit(btc, (self.x-10, self.y-20)) # رمز أمازيغي yaz = font_small.render("ⵣ", True, GOLD) screen.blit(yaz, (self.x-10, self.y+5)) # تأثير لمعان shine = int(5 * math.sin(math.radians(self.angle))) pygame.draw.circle(screen, (255,255,255,100), (center[0]-8, center[1]-8), 6) class Rider: def __init__(self): self.x = 150 self.y = HEIGHT//2 self.vy = 0 def update(self): keys = pygame.key.get_pressed() if keys[pygame.K_SPACE] or keys[pygame.K_UP]: self.vy = -6 self.vy += 0.35 self.y += self.vy self.y = max(50, min(HEIGHT-80, self.y)) def draw(self): # الفارس الريفي مبسط - تقدر تبدلو بصورة الحصان ديالك من بعد pygame.draw.circle(screen, GOLD, (int(self.x), int(self.y)), 25) pygame.draw.rect(screen, RIF_RED, (self.x-15, self.y+20, 30, 20)) # حصان pygame.draw.ellipse(screen, WHITE, (self.x-30, self.y+10, 60, 20)) rider = Rider() coins = [RiffanCoin() for _ in range(5)] running = True while running: for event in pygame.event.get(): if event.type == pygame.QUIT: running = False # خلفية screen.fill((15, 25, 35)) # نجوم for i in range(50): pygame.draw.circle(screen, WHITE, (random.randint(0,WIDTH), random.randint(0,100)), 1) rider.update() for coin in coins: coin.update() coin.draw() # تصادم dist = math.hypot(rider.x - coin.x, rider.y - coin.y) if dist < 45: coins_collected += 1 coin.x = WIDTH + 200 # هنا فين تربط بالبلوكشين من بعد print(f"Collected {coins_collected} {TOKEN_NAME}!") rider.draw() # HUD title = font_big.render("RIFFAN.IO", True, GOLD) screen.blit(title, (20, 10)) sub = font_small.render(f"REPUBLIC OF RIF - 1921", True, RIF_RED) screen.blit(sub, (20, 65)) score_text = font_small.render(f"{TOKEN_NAME}: {coins_collected} / {TOKEN_SUPPLY}", True, WHITE) screen.blit(score_text, (WIDTH-350, 20)) link = font_small.render("riffan.io - Buy on Dex", True, AMAZIGH_BLUE) screen.blit(link, (WIDTH-350, 50)) pygame.display.flip() clock.tick(60) pygame.quit() sys.exit()
